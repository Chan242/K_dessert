



CREATE TABLE MEMBER(         
M_INDEX   NUMBER(6)      ,
M_ID   VARCHAR2(50)   NOT NULL   ,
M_PASSWORD   VARCHAR2(50)   NOT NULL   ,
M_NAME   VARCHAR2(100)   NOT NULL   ,
M_TEL   VARCHAR2(100)   NOT NULL   ,
M_EMAIL   VARCHAR2(100)   NOT NULL   ,
M_ADDRESS   VARCHAR2(100)   NOT NULL   ,
M_ADDRESS_SEC   VARCHAR2(100)   NOT NULL   ,
M_BIRTH   DATE   NOT NULL   ,
M_SIGN_TIME   DATE   DEFAULT SYSDATE   ,
M_CORR_DATE DATE DEFAULT SYSDATE,
M_ADM_CHECK NUMBER(1) DEFAULT 0,
M_NOTE   VARCHAR2(4000)      
);      

ALTER TABLE MEMBER
ADD CONSTRAINT MEMBER_M_INDEX_PK PRIMARY KEY(M_INDEX)
ADD CONSTRAINT MEMBER_M_ID_UK UNIQUE (M_ID)
ADD CONSTRAINT MEMBER_M_EMAIL_UK UNIQUE (M_EMAIL);


CREATE TABLE PRODUCT(
P_INDEX   NUMBER(6),
P_NAME   VARCHAR2(100)   NOT NULL,
P_PRICE   NUMBER(6)   NOT NULL,
P_STOCK   NUMBER(6)   DEFAULT 0,
P_INTRO   CLOB   ,
P_OPEN   NUMBER(1,0)   DEFAULT 0,
P_IMAGE   VARCHAR2(4000),
P_CRE_DATE DATE DEFAULT SYSDATE,
P_CORR_DATE DATE DEFAULT SYSDATE
);

ALTER TABLE PRODUCT
ADD CONSTRAINT PRODUCT_P_INDEX_PK PRIMARY KEY (P_INDEX);


CREATE TABLE DESSERT(
D_INDEX   NUMBER(6)   ,
D_NAME   VARCHAR2(255)   NOT NULL,
D_TEXT   CLOB   ,
D_IMAGE   VARCHAR2(4000),
D_VIEW   NUMBER(6) DEFAULT 0,   
D_OPEN   NUMBER(1,0)   DEFAULT 0
);

ALTER TABLE DESSERT
ADD CONSTRAINT DESSERT_D_INDEX_PK PRIMARY KEY (D_INDEX);

CREATE TABLE FREE_BOARD(
F_index   NUMBER(6)   ,
M_INDEX   NUMBER(6) ,
F_SUBJECT   VARCHAR2(255)   ,
F_TEXT   CLOB   ,
F_IMAGE   VARCHAR2(4000)   ,
F_CRE_DATE   DATE   DEFAULT SYSDATE,
F_VIEW NUMBER(6),
F_NOTICE   NUMBER(1,0)   DEFAULT 0,
F_CORR_DATE DATE DEFAULT SYSDATE
);


ALTER TABLE FREE_BOARD
ADD CONSTRAINT FREEBOARD_F_INDEX_PK PRIMARY KEY (F_INDEX);



CREATE TABLE REPLY(
R_INDEX   NUMBER(6),
M_INDEX   NUMBER(6),
R_TEXT   VARCHAR2(4000)   ,
F_INDEX   NUMBER(6)   ,
R_CRE_DATE   DATE   DEFAULT SYSDATE,
R_CORR_DATE DATE DEFAULT SYSDATE
);

ALTER TABLE REPLY
ADD CONSTRAINT REPLY_R_INDEX_PK PRIMARY KEY(R_INDEX)
ADD CONSTRAINT REPLY_F_INDEX_FK FOREIGN KEY (F_INDEX) REFERENCES FREE_BOARD (F_INDEX) ON DELETE CASCADE;

CREATE TABLE TAG(
T_INDEX   NUMBER(6),
T_NAME   VARCHAR2(20)NOT NULL
);

ALTER TABLE TAG
ADD CONSTRAINT TAG_T_INDEX_PK PRIMARY KEY (T_INDEX)
ADD CONSTRAINT TAG_T_NAME_UK UNIQUE (T_NAME);

CREATE TABLE BASKET(
M_INDEX   NUMBER(6),
P_INDEX   NUMBER(6),
B_STOCK   NUMBER(2),
B_CRE_DATE DATE DEFAULT SYSDATE
);


ALTER TABLE BASKET
ADD CONSTRAINT BASKET_M_INDEX_FK FOREIGN KEY (M_INDEX) REFERENCES MEMBER(M_INDEX) ON DELETE CASCADE
ADD CONSTRAINT BASKET_P_INDEX_FK FOREIGN KEY (P_INDEX) REFERENCES PRODUCT(P_INDEX) ON DELETE CASCADE;


CREATE TABLE PRODUCT_TAG(
P_INDEX   NUMBER(6),
T_NAME VARCHAR(255)
);

ALTER TABLE PRODUCT_TAG
ADD CONSTRAINT PT_P_INDEX_FK FOREIGN KEY(P_INDEX) REFERENCES PRODUCT (P_INDEX)  ON DELETE CASCADE;





CREATE TABLE STATUS(
STA_INDEX   NUMBER(2),
STA_STATUS   VARCHAR2(255) NOT NULL,
STA_NOTICE   VARCHAR2(255) NOT NULL
);

ALTER TABLE STATUS
ADD CONSTRAINT STATUS_STA_INDEX_PK PRIMARY KEY (STA_INDEX)
ADD CONSTRAINT STATUS_STA_STATUS_UK UNIQUE (STA_STATUS)
ADD CONSTRAINT STATUS_STA_NOTICE_UK UNIQUE (STA_NOTICE);

CREATE TABLE S_ORDER(
O_INDEX   NUMBER(6),
STA_STATUS   VARCHAR2(255),
O_TIME   DATE   DEFAULT SYSDATE,
O_TOTAL NUMBER(7),
M_INDEX NUMBER(6),
O_NAME VARCHAR2(100),
O_ADDRESS   VARCHAR2(100)   ,
O_ADDRESS_SEC   VARCHAR2(100)   ,
O_TEL   VARCHAR2(100)   
);

ALTER TABLE S_ORDER
ADD CONSTRAINT SO_O_INDEX PRIMARY KEY(O_INDEX);


create table product_order(

O_INDEX	NUMBER(6),
P_INDEX	NUMBER(6),
P_NAME VARCHAR2(255),
PO_STOCK NUMBER(2),
PO_PRICE NUMBER(6)
);

CREATE TABLE POINT(
POINT_INDEX NUMBER(6),
M_INDEX NUMBER(6),
POINT_DATE DATE DEFAULT SYSDATE,
POINT_POINT NUMBER(9)
);

CREATE TABLE EVENT(
E_INDEX   NUMBER(6),
E_NAME   VARCHAR2(100)   NOT NULL,
E_IMAGE   VARCHAR2(4000),
E_EVENT_DATE    DATE DEFAULT SYSDATE NOT NULL,
E_EXPLAIN    VARCHAR2(4000),
E_OPEN   NUMBER(1,0)   DEFAULT 0 NOT NULL,
E_CRE_DATE DATE DEFAULT SYSDATE NOT NULL,
E_CORR_DATE DATE DEFAULT SYSDATE NOT NULL,
E_NOTE  VARCHAR2(4000)
);

ALTER TABLE EVENT
ADD CONSTRAINT EVENT_E_INDEX_PK PRIMARY KEY (E_INDEX);
